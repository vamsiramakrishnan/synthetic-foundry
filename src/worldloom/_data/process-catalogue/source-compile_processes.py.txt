def load(path):
    return json.load(open(path))


def activities_for(cat, industry):
    ov = cat["industry_overlays"][industry]
    streams = {}
    for sid, s in cat["value_streams"].items():
        streams[sid] = {"name": s["name"], "kind": "universal", "core": sid in ov["core_streams"], "activities": s["activities"], "calibrate": s.get("calibrate", [])}
    for sid, s in ov["specific"].items():
        streams[sid] = {"name": s["name"], "kind": "industry_specific", "core": True, "activities": s["activities"], "calibrate": s.get("calibrate", [])}
    return streams


def owner_kind(cat, model, function):
    rules = cat["operating_models"][model]
    for kind, fams in rules.items():
        if function in fams:
            return kind
    return "group_function"


def bind_owner(kind, bus):
    if kind == "BU":
        return [b for b in bus if b["archetype"] in ("product_line", "geography", "customer_segment", "channel", "legal_entity")] or bus
    if kind == "shared_service":
        return [b for b in bus if b["archetype"] == "shared_service_centre"] or [b for b in bus if b["archetype"] == "group_function"] or bus[:1]
    return [b for b in bus if b["archetype"] == "group_function"] or bus[:1]


def compile_company(cat, spec):
    industry = spec["industry"]
    landscape = {**DEFAULT_LANDSCAPE, **spec.get("landscape", {})}
    streams = activities_for(cat, industry)
    rows, cov = [], Counter()
    for sid, s in streams.items():
        for act in s["activities"]:
            a = dict(zip(COLS, act))
            kind = owner_kind(cat, spec["operating_model"], a["function"])
            owners = bind_owner(kind, spec["bus"])
            prod = landscape.get(a["sor_class"], "unbound")
            objects = list(cat["sor_classes"].get(a["sor_class"], {}).get("products", {}).get(prod, {}).keys())
            prior = cat["channel_priors"][a["type"]]
            for bu in owners:
                for country in spec["countries"]:
                    variant = {}
                    rv = cat["regional_variants"].get(country, {})
                    for tag in a["tags"]:
                        for k in cat["variant_tags"].get(tag, []):
                            if k in rv:
                                variant[k] = rv[k]
                    rows.append({"activity_id": a["id"], "stream": sid, "stream_name": s["name"], "activity": a["name"], "apqc": a["apqc"],
                                 "function": a["function"], "owner_kind": kind, "owner_bu": bu["name"], "bu_archetype": bu["archetype"], "country": country,
                                 "sor_class": a["sor_class"], "sor_product": prod, "sor_objects": objects, "type": a["type"],
                                 "channels": [c for c, p in prior.items() if p >= 0.5], "channels_optional": [c for c, p in prior.items() if 0 < p < 0.5],
                                 "control": a["control"], "exception": a["exception"], "variant": variant,
                                 "eval_templates": [t["id"] for t in cat["eval_templates"]]})
            status = "corpus_calibrated" if s["calibrate"] else ("overlay" if s["kind"] == "industry_specific" else "backbone")
            cov[(industry, sid, status)] += 1
    return rows, cov


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--catalogue", default="catalogue.json"); ap.add_argument("--spec"); ap.add_argument("--all", action="store_true"); ap.add_argument("--out", default="compiled")
    a = ap.parse_args()
    cat = load(a.catalogue)
    os.makedirs(a.out, exist_ok=True)
    specs = []
    if a.spec:
        specs.append(load(a.spec))
    if a.all:
        for ind, org in DEFAULT_ORGS.items():
            specs.append({"name": f"default-{ind}", "industry": ind, **org})
    total_cov = Counter(); summary = []
    for spec in specs:
        rows, cov = compile_company(cat, spec)
        total_cov.update(cov)
        name = spec.get("name", spec["industry"])
        with open(os.path.join(a.out, f"{name}.jsonl"), "w") as f:
            for r in rows:
                f.write(json.dumps(r, ensure_ascii=False) + "\n")
        n_act = len({r["activity_id"] for r in rows})
        summary.append({"company": name, "industry": spec["industry"], "operating_model": spec["operating_model"], "bus": len(spec["bus"]), "countries": len(spec["countries"]),
                        "distinct_activities": n_act, "activity_instances": len(rows), "streams": len({r["stream"] for r in rows}),
                        "eval_demands": len(rows) * len(cat["eval_templates"]), "unbound_systems": sorted({r["sor_class"] for r in rows if r["sor_product"] == "unbound"})})
    matrix = defaultdict(dict)
    for (ind, sid, status), n in total_cov.items():
        matrix[ind][sid] = {"status": status, "activities": n}
    json.dump({"summary": summary, "coverage": matrix}, open(os.path.join(a.out, "coverage.json"), "w"), indent=1)
    with open(os.path.join(a.out, "coverage.csv"), "w") as f:
        f.write("industry,stream,status,activities\n")
        for ind in sorted(matrix):
            for sid, v in sorted(matrix[ind].items()):
                f.write(f"{ind},{sid},{v['status']},{v['activities']}\n")
    for s in summary:
        print(json.dumps(s))
    inds = len(matrix); streams = sum(len(v) for v in matrix.values()); statuses = Counter(v["status"] for m in matrix.values() for v in m.values())
    print(f"coverage: {inds} industries, {streams} industry-stream cells, {dict(statuses)}")


if __name__ == "__main__":
    main()
